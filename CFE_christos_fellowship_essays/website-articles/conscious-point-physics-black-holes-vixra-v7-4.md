import numpy as np

N         = 1e61                     # Approximate bit count for solar-mass object
G         = 6.67430e-11              # m^3 kg^-1 s^-2
M         = 1.989e30                 # Solar mass (kg)
c         = 3e8                      # m/s
ell_p     = 1.616255e-35             # Planck length (m)
delta     = ell_p / N                # Discrete cutoff
n_events  = 100000                   # Monte Carlo samples

def horizon_radius(M):
    base         = 2 * G * M / c**2
    fluctuation  = np.random.normal(1.0, 0.001)  # Bit-statistics noise
    return base * fluctuation

ensemble = [horizon_radius(M) for _ in range(n_events)]
rs_mean  = np.mean(ensemble) / 1000.0    # Convert to km

print(f"Mean Schwarzschild radius r_s: {rs_mean:.2f} km")

A = 4 * np.pi * (rs_mean * 1000.0)**2
S = A / (4 * ell_p**2)
